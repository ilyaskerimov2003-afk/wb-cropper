    <script src="https://cdnjs.cloudflare.com/ajax/libs/cropperjs/1.5.13/cropper.min.js"></script>

    <script src="https://telegram.org/js/telegram-web-app.js"></script>

        body { background: #000; color: #fff; font-family: sans-serif; margin: 0; padding: 0; }

        button { border: none; padding: 12px; border-radius: 10px; font-weight: bold; font-size: 14px; cursor: pointer; }

        .btn-next { background: #007bff; color: white; grid-column: span 2; }

        .btn-border { background: #f8f9fa; color: #000; }

        .btn-rotate { background: #444; color: #fff; }

        .info { text-align: center; color: #aaa; font-size: 13px; grid-column: span 2; margin-bottom: 5px; }

<body>

    <div class="container">

        <div id="image-wrapper"><img id="target-image"></div>

        <div class="controls">

            <div id="counter" class="info">Загрузка фото...</div>

            <button class="btn-border" onclick="setFullImage()">⚪ С полями</button>

            <button class="btn-rotate" onclick="rotateCrop()">🔄 Повернуть</button>

            <button class="btn-next" id="next-btn" onclick="nextPhoto()">Далее</button>

        </div>

    </div>


    <script>

        const tg = window.Telegram.WebApp;

        tg.expand(); 


        const urlParams = new URLSearchParams(window.location.search);

        const photos = JSON.parse(urlParams.get('photos') || '[]');

        const format = urlParams.get('format') || '10x15';

        

        let currentIndex = 0;

        let cropper;

        let currentAspect = (format === '10x15' || format === '4x6 Полароид') ? 2/3 : 3/4;

        const results = [];

        const img = document.getElementById('target-image');


        function initCropper() {

            if (cropper) cropper.destroy();

            // Если фото альбомное, переворачиваем рамку сразу

            let aspect = currentAspect;

            if (img.naturalWidth > img.naturalHeight) aspect = 1 / aspect;


            cropper = new Cropper(img, {

                aspectRatio: aspect,

                viewMode: 1,

                dragMode: 'move',

                autoCropArea: 1,

                background: false

            });

        }


        function loadPhoto(index) {

            if (index >= photos.length) {

                tg.sendData(JSON.stringify(results));

                return;

            }

            document.getElementById('counter').innerText = `Фото ${index + 1} из ${photos.length}`;

            img.src = photos[index];

            img.onload = initCropper;

            if (index === photos.length - 1) document.getElementById('next-btn').innerText = "Завершить";

        }


        function setFullImage() {

            results.push({ index: currentIndex, is_full: true });

            currentIndex++;

            loadPhoto(currentIndex);

        }


        function rotateCrop() {

            const newAspect = 1 / cropper.options.aspectRatio;

            cropper.setAspectRatio(newAspect);

        }


        function nextPhoto() {

            const d = cropper.getData();

            results.push({ 

                index: currentIndex, 

                x: Math.round(d.x), y: Math.round(d.y), 

                w: Math.round(d.width), h: Math.round(d.height), 

                is_full: false 

            });

            currentIndex++;

            loadPhoto(currentIndex);

        }


        loadPhoto(0);

    </script>
